        function handleStartExam(e) {
            e.preventDefault();
            const nama = document.getElementById('input-nama').value.trim();
            const kelas = document.getElementById('input-kelas').value;

            if (!nama || !kelas) {
                alert("Silakan isi nama dan pilih kelas terlebih dahulu!");
                return;
            }

            let materi = "Teks Deskripsi";
            if (kelas === "7A") {
                materi = "Teks Deskripsi";
                currentQuestions = buildDeskripsiQuestions();
            } else {
                materi = "Teks Laporan Hasil Observasi (LHO)";
                currentQuestions = buildLhoQuestions();
            }

            currentUser = {
                nama: nama,
                kelas: kelas,
                materi: materi,
                answers: {},
                remainingSeconds: EXAM_DURATION_SECONDS,
                date: new Date().toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
            };

            document.getElementById('display-exam-nama').innerText = currentUser.nama;
            document.getElementById('display-exam-kelas').innerText = `Kelas ${currentUser.kelas} • ${currentUser.materi}`;
            document.getElementById('display-avatar-class').innerText = currentUser.kelas;

            isExamActive = true;
            switchTab('exam');
            currentQuestionIndex = 0;
            renderQuestion();
            renderQuestionGrid();
            startTimer();
        }

        function startTimer() {
            clearInterval(timerInterval);
            const timerDisplay = document.getElementById('exam-timer');

            timerInterval = setInterval(() => {
                if (currentUser.remainingSeconds <= 0) {
                    clearInterval(timerInterval);
                    alert("Waktu ujian telah habis! Jawaban Anda disubmit otomatis.");
                    submitExam();
                    return;
                }

                currentUser.remainingSeconds--;
                const hrs = Math.floor(currentUser.remainingSeconds / 3600);
                const mins = Math.floor((currentUser.remainingSeconds % 3600) / 60);
                const secs = currentUser.remainingSeconds % 60;

                timerDisplay.innerText = 
                    `${hrs.toString().padStart(2, '0')}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
            }, 1000);
        }

        function renderQuestion() {
            const q = currentQuestions[currentQuestionIndex];
            document.getElementById('q-number').innerText = currentQuestionIndex + 1;

            const typeLabels = {
                "PG": "Pilihan Ganda",
                "PGK": "PG Kompleks (Multi-Jawaban)",
                "BS": "Benar / Salah",
                "ISIAN": "Isian Singkat",
                "URAIAN": "Uraian / Essay"
            };
            document.getElementById('q-type-badge').innerText = typeLabels[q.type] || q.type;
            document.getElementById('q-score-weight').innerText = q.type === 'URAIAN' ? "Bobot: 2 Poin" : "Bobot: 1 Poin";

            document.getElementById('q-text-container').innerHTML = `<p>${q.question}</p>`;

            const optsContainer = document.getElementById('q-options-container');
            optsContainer.innerHTML = '';

            const currentAns = currentUser.answers[q.id];

            if (q.type === 'PG') {
                q.options.forEach(opt => {
                    const optKey = opt.charAt(0);
                    const isChecked = currentAns === optKey;
                    optsContainer.innerHTML += `
                        <label class="flex items-start gap-3 p-3 rounded-xl border border-slate-200 hover:border-teal-500 hover:bg-teal-50/50 cursor-pointer transition ${isChecked ? 'bg-teal-50 border-teal-500 font-medium' : 'bg-white'}">
                            <input type="radio" name="q_opt" value="${optKey}" ${isChecked ? 'checked' : ''} onchange="saveAnswer('${q.id}', '${optKey}')" class="mt-1 text-teal-600 focus:ring-teal-500">
                            <span class="text-sm text-slate-800">${opt}</span>
                        </label>
                    `;
                });
            } else if (q.type === 'PGK') {
                const selectedList = Array.isArray(currentAns) ? currentAns : [];
                q.options.forEach(opt => {
                    const isChecked = selectedList.includes(opt.id);
                    optsContainer.innerHTML += `
                        <label class="flex items-start gap-3 p-3 rounded-xl border border-slate-200 hover:border-teal-500 hover:bg-teal-50/50 cursor-pointer transition ${isChecked ? 'bg-teal-50 border-teal-500 font-medium' : 'bg-white'}">
                            <input type="checkbox" value="${opt.id}" ${isChecked ? 'checked' : ''} onchange="savePgkAnswer(${q.id}, '${opt.id}')" class="mt-1 text-teal-600 rounded focus:ring-teal-500">
                            <span class="text-sm text-slate-800"><strong>[${opt.id}]</strong> ${opt.text}</span>
                        </label>
                    `;
                });
            } else if (q.type === 'BS') {
                ['BENAR', 'SALAH'].forEach(val => {
                    const isChecked = currentAns === val;
                    optsContainer.innerHTML += `
                        <label class="flex items-center gap-3 p-3.5 rounded-xl border border-slate-200 hover:border-teal-500 hover:bg-teal-50/50 cursor-pointer transition ${isChecked ? 'bg-teal-50 border-teal-500 font-bold' : 'bg-white'}">

                        </label>
                    `;
                });
            } else if (q.type === 'ISIAN') {
                const valText = currentAns || '';
                optsContainer.innerHTML = `
                    <div class="mt-2">
                        <label class="block text-xs text-slate-500 mb-1">Tuliskan jawaban singkat Anda:</label>

        function saveAnswer(qId, val) {
            currentUser.answers[qId] = val;
            updateAnsweredBadge();
            renderQuestionGrid();
        }

        function savePgkAnswer(qId, optId) {
            let currentList = currentUser.answers[qId] || [];
            if (!Array.isArray(currentList)) currentList = [];

            if (currentList.includes(optId)) {
                currentList = currentList.filter(x => x !== optId);
            } else {
                currentList.push(optId);
            }

            currentUser.answers[qId] = currentList;
            updateAnsweredBadge();
            renderQuestionGrid();
        }

        function navigateQuestion(step) {
            const nextIdx = currentQuestionIndex + step;
            if (nextIdx >= 0 && nextIdx < currentQuestions.length) {
                currentQuestionIndex = nextIdx;
                renderQuestion();
            }
        }

        function jumpToQuestion(idx) {
            currentQuestionIndex = idx;
            renderQuestion();
        }

        function updateAnsweredBadge() {
            const count = Object.keys(currentUser.answers).filter(k => {
                const val = currentUser.answers[k];
                return val !== undefined && val !== '' && !(Array.isArray(val) && val.length === 0);
            }).length;
            document.getElementById('answered-count-badge').innerText = `${count}/100`;
        }

        function renderQuestionGrid() {
            const gridContainer = document.getElementById('q-grid-container');
            const filterType = document.getElementById('filter-q-type').value;
            gridContainer.innerHTML = '';

            currentQuestions.forEach((q, idx) => {
                if (filterType !== 'ALL' && q.type !== filterType) return;

                const ans = currentUser.answers[q.id];
                const isAnswered = ans !== undefined && ans !== '' && !(Array.isArray(ans) && ans.length === 0);
                const isActive = idx === currentQuestionIndex;

                let bgClass = "bg-slate-100 text-slate-700 hover:bg-slate-200";
                if (isAnswered) bgClass = "bg-emerald-500 text-white font-bold";
                if (isActive) bgClass = "bg-teal-600 text-white font-bold ring-2 ring-teal-300 ring-offset-1";

                gridContainer.innerHTML += `
                    <button onclick="jumpToQuestion(${idx})" class="w-full h-8 text-xs rounded-lg flex items-center justify-center transition ${bgClass}">
                        ${idx + 1}
                    </button>
                `;
            });
        }

        function confirmSubmitExam() {
            const answeredCount = Object.keys(currentUser.answers).filter(k => {
                const val = currentUser.answers[k];
                return val !== undefined && val !== '' && !(Array.isArray(val) && val.length === 0);
            }).length;

            if (confirm(`Apakah Anda yakin ingin menyelesaikan ujian?\n\nSoal dijawab: ${answeredCount} / 100`)) {
                submitExam();
            }
        }

        function submitExam() {
            clearInterval(timerInterval);
            isExamActive = false;

            let pgScore = 0;
            let pgkScore = 0;
            let bsScore = 0;
            let isianScore = 0;
            let uraianScore = 0;

            currentQuestions.forEach(q => {
                const userAns = currentUser.answers[q.id];

                if (q.type === 'PG') {
                    if (userAns === q.key) pgScore += 1;
                } else if (q.type === 'PGK') {
                    if (Array.isArray(userAns)) {
                        const isExact = q.keys.length === userAns.length && q.keys.every(k => userAns.includes(k));
                        if (isExact) pgkScore += 1;
                    }
                } else if (q.type === 'BS') {
                    if (userAns === q.key) bsScore += 1;
                } else if (q.type === 'ISIAN') {
                    if (typeof userAns === 'string' && userAns.trim() !== '') {
                        const cleanUser = userAns.trim().toLowerCase();
                        const matched = q.key.some(k => cleanUser.includes(k.toLowerCase()));
                        if (matched) isianScore += 1;
                    }
                } else if (q.type === 'URAIAN') {
                    if (typeof userAns === 'string' && userAns.trim().length > 10) {
                        const cleanUser = userAns.trim().toLowerCase();
                        const keys = q.sampleKey.split(',').map(k => k.trim().toLowerCase());
                        const matchCount = keys.filter(k => cleanUser.includes(k)).length;
                        if (matchCount >= 2) uraianScore += 2;
                        else if (matchCount >= 1) uraianScore += 1;
                    }
                }
            });

            const totalScore = pgScore + pgkScore + bsScore + isianScore + uraianScore;
            const durationUsed = EXAM_DURATION_SECONDS - currentUser.remainingSeconds;
            const durationMin = Math.ceil(durationUsed / 60);

            const record = {
                id: Date.now(),
                nama: currentUser.nama,
                kelas: currentUser.kelas,
                materi: currentUser.materi,
                scorePG: pgScore,
                scorePGK: pgkScore,
                scoreBS: bsScore,
                scoreIsian: isianScore,
                scoreUraian: uraianScore,
                totalScore: totalScore,
                durationMinutes: durationMin,
                date: currentUser.date,
                userAnswers: currentUser.answers
            };

            saveResultToStorage(record);
            displayResultPage(record);
            switchTab('result');
        }

        function displayResultPage(record) {
            document.getElementById('result-student-info').innerText = `${record.nama} • Kelas ${record.kelas} (${record.materi})`;
            document.getElementById('result-total-score').innerText = record.totalScore.toFixed(1);

            document.getElementById('score-pg').innerText = `${record.scorePG}/50`;
            document.getElementById('score-pgk').innerText = `${record.scorePGK}/10`;
            document.getElementById('score-bs').innerText = `${record.scoreBS}/10`;
            document.getElementById('score-isian').innerText = `${record.scoreIsian}/10`;
            document.getElementById('score-uraian').innerText = `${record.scoreUraian}/20`;

            let grade = "Cukup";
            if (record.totalScore >= 90) grade = "Sangat Baik (A)";
            else if (record.totalScore >= 80) grade = "Baik (B)";
            else if (record.totalScore >= 70) grade = "Cukup Baik (C)";
            else grade = "Perlu Bimbingan (D)";

            document.getElementById('result-grade-label').innerText = `Predikat: ${grade}`;

            renderDiscussionList(record.userAnswers);
        }

        function renderDiscussionList(userAnsMap) {
            const discList = document.getElementById('discussion-list');
            discList.innerHTML = '';

            currentQuestions.forEach((q, idx) => {
                const uAns = userAnsMap[q.id] || "<i>Belum Dijawab</i>";
                let keyStr = "";

                if (q.type === 'PG') keyStr = q.key;
                else if (q.type === 'PGK') keyStr = q.keys.join(', ');
                else if (q.type === 'BS') keyStr = q.key;
                else if (q.type === 'ISIAN') keyStr = q.key.join(' / ');
                else if (q.type === 'URAIAN') keyStr = q.rubrik;

                discList.innerHTML += `
                    <div class="p-4 rounded-xl border border-slate-200 bg-white space-y-2 text-xs sm:text-sm">
                        <div class="font-bold text-slate-800 flex justify-between">
                            <span>Soal #${idx + 1} (${q.type})</span>
                        </div>
                        <p class="text-slate-700">${q.question}</p>
                        <div class="p-2.5 bg-slate-50 rounded-lg border border-slate-200 space-y-1">
                            <div><strong class="text-indigo-600">Jawaban Anda:</strong> ${Array.isArray(uAns) ? uAns.join(', ') : uAns}</div>
                            <div><strong class="text-emerald-600">Kunci/Rubrik:</strong> ${keyStr}</div>
                            <div class="text-xs text-slate-500 mt-1"><strong>Pembahasan:</strong> ${q.discussion}</div>
                        </div>
                    </div>
                `;
            });
        }

        function renderLeaderboard(filterClass = 'ALL') {
            const results = getStoredResults();

            let filtered = results;
            if (filterClass !== 'ALL') {
                filtered = results.filter(r => r.kelas === filterClass);
            }

            filtered.sort((a, b) => b.totalScore - a.totalScore || a.durationMinutes - b.durationMinutes);

            const top10 = filtered.slice(0, 10);
            const tbody = document.getElementById('leaderboard-tbody');
            tbody.innerHTML = '';

            if (top10.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="p-4 text-center text-slate-400">Belum ada data nilai pengerjaan ujian.</td></tr>`;
                return;
            }

            top10.forEach((r, idx) => {
                let badgeClass = "bg-slate-100 text-slate-600";
                if (idx === 0) badgeClass = "bg-amber-100 text-amber-700 font-bold border border-amber-300";
                if (idx === 1) badgeClass = "bg-slate-200 text-slate-700 font-bold border border-slate-300";
                if (idx === 2) badgeClass = "bg-orange-100 text-orange-700 font-bold border border-orange-300";

                tbody.innerHTML += `
                    <tr class="hover:bg-slate-50">
                        <td class="p-3 text-center">
                            <span class="inline-block w-6 h-6 rounded-full text-xs flex items-center justify-center mx-auto ${badgeClass}">
                                ${idx + 1}
                            </span>
                        </td>
                        <td class="p-3 font-semibold text-slate-800">${r.nama}</td>
                        <td class="p-3 text-center"><span class="px-2 py-0.5 rounded bg-slate-100 font-bold">${r.kelas}</span></td>
                        <td class="p-3 text-center text-slate-600 text-[11px]">${r.materi}</td>
                        <td class="p-3 text-center font-extrabold text-teal-600 text-sm">${r.totalScore.toFixed(1)}</td>
                        <td class="p-3 text-center text-slate-500">${r.durationMinutes} Menit</td>
                        <td class="p-3 text-right text-slate-400 text-[11px]">${r.date}</td>
                    </tr>
                `;
            });
        }

        function filterLeaderboard(cls) {
            document.querySelectorAll('.lb-tab-btn').forEach(btn => {
                btn.classList.remove('border-teal-600', 'text-teal-700');

        }

        function renderGuruTable() {
            const results = getStoredResults();
            const filterCls = document.getElementById('guru-filter-kelas').value;

            let filtered = results;
            if (filterCls !== 'ALL') {
                filtered = results.filter(r => r.kelas === filterCls);
            }

            const tbody = document.getElementById('guru-table-tbody');
            tbody.innerHTML = '';

            if (filtered.length === 0) {
                tbody.innerHTML = `<tr><td colspan="12" class="p-4 text-center text-slate-400">Belum ada siswa yang mengerjakan ujian.</td></tr>`;
                return;
            }

            filtered.forEach((r, idx) => {

                        <td class="p-3">${idx + 1}</td>
                        <td class="p-3 font-semibold text-slate-800">${r.nama}</td>
                        <td class="p-3 text-center"><span class="px-2 py-0.5 rounded bg-slate-100 font-bold">${r.kelas}</span></td>
                        <td class="p-3 text-center text-[10px] text-slate-600">${r.materi}</td>
                        <td class="p-3 text-center">${r.scorePG}</td>
                        <td class="p-3 text-center">${r.scorePGK}</td>
                        <td class="p-3 text-center">${r.scoreBS}</td>
                        <td class="p-3 text-center">${r.scoreIsian}</td>
                        <td class="p-3 text-center">${r.scoreUraian}</td>
                        <td class="p-3 text-center font-bold text-teal-600">${r.totalScore.toFixed(1)}</td>
                        <td class="p-3 text-right text-slate-400 text-[10px]">${r.date}</td>
                        <td class="p-3 text-center">
                            <button onclick="deleteResultRecord(${r.id})" class="p-1 text-rose-600 hover:text-rose-800" title="Hapus Data">
                                <i class="fa-solid fa-trash"></i>
                            </button>
                        </td>
                    </tr>
                `;
            });
        }

        function deleteResultRecord(recordId) {
            if (confirm("Hapus data nilai siswa ini?")) {
                let results = getStoredResults();
                results = results.filter(r => r.id !== recordId);
                localStorage.setItem('bahasa_exam_results', JSON.stringify(results));
                renderGuruTable();
            }
        }

        function clearAllData() {
            if (confirm("Apakah Anda yakin ingin MENGHAPUS SELURUH data nilai dari sistem?")) {
                localStorage.removeItem('bahasa_exam_results');
                renderGuruTable();
            }
        }

        function exportToExcel() {
            const results = getStoredResults();
            if (results.length === 0) {
                alert("Tidak ada data nilai untuk diekspor!");
                return;
            }

            const excelData = results.map((r, index) => ({
                "No": index + 1,
                "Nama Siswa": r.nama,
                "Kelas": r.kelas,
                "Materi Ujian": r.materi,
                "Skor PG (Max 50)": r.scorePG,
                "Skor PGK (Max 10)": r.scorePGK,
                "Skor Benar/Salah (Max 10)": r.scoreBS,
                "Skor Isian (Max 10)": r.scoreIsian,
                "Skor Uraian (Max 20)": r.scoreUraian,
                "Nilai Akhir": r.totalScore,
                "Waktu (Menit)": r.durationMinutes,
                "Tanggal Ujian": r.date
            }));

            const worksheet = XLSX.utils.json_to_sheet(excelData);
            const workbook = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(workbook, worksheet, "Laporan Nilai Bahasa Indonesia");

            worksheet["!cols"] = [{ wch: 5 }, { wch: 25 }, { wch: 10 }, { wch: 30 }];

            XLSX.writeFile(workbook, `Laporan_Nilai_Ujian_${new Date().toISOString().slice(0,10)}.xlsx`);
        }
